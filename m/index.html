const USERNAME = 'gtalex1'; const REPO = 'recursos'; const BASE_FOLDER = 'm'; const catalogContainer = document.getElementById('catalog-container'); const loading = document.getElementById('loading'); let globalIndex = 0; // Función para consultar la API de GitHub async function fetchDirectory(path) { try { const response = await fetch(`https://api.github.com/repos/${USERNAME}/${REPO}/contents/${path}`); if (!response.ok) return []; return await response.json(); } catch (e) { console.error("Error leyendo directorio:", path, e); return []; } } // Función RECURSIVA para buscar todas las imágenes PNG sin importar la profundidad async function scanFolderRecursively(path) { const contents = await fetchDirectory(path); let pngFiles = []; for (const item of contents) { if (item.type === 'file' && item.name.toLowerCase().endsWith('.png')) { pngFiles.push(item); } else if (item.type === 'dir') { // Si es una carpeta, entrar y buscar dentro de ella const subFiles = await scanFolderRecursively(item.path); pngFiles = pngFiles.concat(subFiles); } } return pngFiles; } async function initCatalog() { try { const rootContents = await fetchDirectory(BASE_FOLDER); if (loading) loading.style.display = 'none'; // Detectar carpetas principales (ej: Animes, Juegos) const mainCategories = rootContents.filter(item => item.type === 'dir'); for (const categoryDir of mainCategories) { // Escanear recursivamente todos los PNG dentro de esta categoría const allSkins = await scanFolderRecursively(categoryDir.path); if (allSkins.length > 0) { renderCategoryGroup(categoryDir.name, allSkins); } } } catch (error) { console.error(error); if (loading) loading.textContent = 'Error al cargar el catálogo de skins.'; } } function renderCategoryGroup(categoryName, imageFiles) { // Título principal de la categoría (Animes, Juegos, etc.) const mainTitle = document.createElement('h2'); mainTitle.className = 'category-title'; mainTitle.textContent = categoryName; catalogContainer.appendChild(mainTitle); const grid = document.createElement('div'); grid.className = 'skins-grid'; catalogContainer.appendChild(grid); imageFiles.forEach(file => { globalIndex++; const skinName = file.name.replace(/\.png$/i, '').replace(/[-_]/g, ' '); const rawUrl = file.download_url; const canvasId = `skin_canvas_${globalIndex}`; const selectId = `cmd_type_${globalIndex}`; // Detectar si la ruta contiene Chica, Slim o Alex para definir el modelo 3D const pathLower = file.path.toLowerCase(); const isSlim = pathLower.includes('chica') || pathLower.includes('slim') || pathLower.includes('alex'); const modelType = isSlim ? 'slim' : 'classic'; // Extraer subcategoría (ej: "Darling in the FranXX") para mostrarlo como etiqueta opcional const pathParts = file.path.split('/'); const subCat = pathParts.length > 3 ? pathParts[2] : ''; const card = document.createElement('div'); card.className = 'skin-card'; card.innerHTML = `

${skinName}

${subCat ? `${subCat} (${modelType})` : ''}
Descargar
`; grid.appendChild(card); // Renderizado 3D de la skin try { const viewer = new skinview3d.SkinViewer({ canvas: document.getElementById(canvasId), width: 180, height: 220, skin: rawUrl, model: modelType }); viewer.animation = new skinview3d.WalkingAnimation(); viewer.animation.speed = 0.5; viewer.autoRotate = true; viewer.autoRotateSpeed = 0.8; } catch (e) { console.error("Error al cargar visor 3D para " + file.name, e); } }); } function copyCustomCommand(url, selectId, modelType, btnElement) { const mode = document.getElementById(selectId).value; let textToCopy = url; if (mode === 'plugin') { textToCopy = `/skin url ${url}`; } else if (mode === 'mod') { textToCopy = `/skin set web ${modelType} "${url}"`; } navigator.clipboard.writeText(textToCopy).then(() => { const originalText = btnElement.textContent; btnElement.textContent = '¡Comando copiado!'; btnElement.style.background = 'var(--accent-amber, #ff9900)'; setTimeout(() => { btnElement.textContent = originalText; btnElement.style.background = ''; }, 1800); }); } // Inicializar el catálogo initCatalog();
